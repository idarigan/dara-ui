## Features

- ✅ localStorage persistence
- ✅ Icon-only mode
- ✅ Fixed width support
- ✅ Updates lang and dir attributes
- ✅ Controlled and uncontrolled modes
- ✅ Custom language list support

---

## Usage

```tsx
import { LanguageChanger, I18nProvider } from "dara-ui";
import { translations } from "dara-ui/translations";

<I18nProvider translations={translations} defaultLanguage="en">
  <LanguageChanger />
</I18nProvider>;
```
